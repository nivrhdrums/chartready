function rcConnect(){
  clearTimeout(rcTimer); if(rcStopped) return;
  let ws;
  try{ ws=new WebSocket((location.protocol==='https:'?'wss://':'ws://')+location.host+'/ws'); }catch(e){ rcStatus('השליטה מרחוק עובדת רק בגרסה שמותקנת על השרת שלך.'); return; }
  rcWs=ws;
  ws.onopen=()=>{ rcRetry=0;
    if(rcPending&&rcPending.auto){ ws.send(JSON.stringify({t:'remoteAuto',name:/iPhone/.test(navigator.userAgent)?'iPhone':/Android/.test(navigator.userAgent)?'Android':'טלפון'})); return; }
    if(rcPending&&rcPending.join){ ws.send(JSON.stringify({t:'pairJoin',code:rcPending.join,name:/iPhone/.test(navigator.userAgent)?'iPhone':/Android/.test(navigator.userAgent)?'Android':'טלפון'})); return; }
    if(RCS.role==='host') ws.send(JSON.stringify({t:'host',token:RCS.token,pass:rcPass||''}));
    else if(RCS.role==='remote'&&RCS.token) ws.send(JSON.stringify({t:'remote',token:RCS.token}));
  };
  ws.onmessage=e=>{ let m; try{ m=JSON.parse(e.data); }catch(x){ return; } rcHandle(m); };
  ws.onclose=()=>{ if(rcWs!==ws) return; RC.connected=false; RC.peer=false; rcMode(false); rcUI();
    if(RCS.role&&(RCS.token||rcPending)&&!rcStopped){ rcRetry=Math.min(rcRetry+1,6); rcTimer=setTimeout(rcConnect,1000*rcRetry); } };
  ws.onerror=()=>{ if(location.hostname.endsWith('claude.ai')||location.protocol==='file:') rcStatus('השליטה מרחוק עובדת רק בגרסה שמותקנת על השרת שלך.'); };
}
function rcHandle(m){
  if(m.t==='hostOk'){ RCS.role='host'; RCS.token=m.token; rcPass=null; rcSave(); RC.connected=true; if((rcPending&&rcPending.pair)||(!rcLocal&&!$('remPairBox').hidden)){ rcPending=null; rcWs.send(JSON.stringify({t:'pairStart'})); } rcUI(); return; } // after a server restart the old code is gone: show a fresh one
  if(m.t==='pairCode'){ showPair(m.code); return; }
  if(m.t==='paired'){ RCS.role='remote'; RCS.token=m.token; RCS.unlocked=true; rcPending=null; rcSave(); RC.connected=true; toast('הטלפון חובר למחשב'); rcUI(); return; }
  if(m.t==='peer'){ RC.connected=true; RC.peer=m.online; if(m.phoneName!=null) RC.phoneName=m.phoneName; if(m.paired!=null) RC.paired=m.paired;
    if(RCS.role==='host'){ if(m.online){ $('remPairBox').hidden=true; hostSendState(true); hostSendLive(true); } }
    else rcMode(m.online);
    rcUI(); return; }
  if(m.t==='revoked'&&rcLocal&&RCS.role==='remote'&&m.why!=='other'){ RCS.role=null; RCS.token=null; rcSave(); rcPending={auto:true}; setTimeout(rcConnect,500); return; }
  if(m.t==='err'&&m.code==='nohost'){ rcStatus('המחשב עוד לא פתח את האפליקציה. מנסה שוב…'); clearTimeout(rcTimer); rcTimer=setTimeout(()=>{ rcPending={auto:true}; rcConnect(); },3000); return; }
  if(m.t==='revoked'){ if(RCS.role==='remote'){ RCS.role=null; RCS.token=null; toast('השיוך למחשב בוטל'); } else { RC.paired=false; } rcSave(); rcMode(false); RC.peer=false; rcUI(); return; }
  if(m.t==='replaced'){ rcStopped=true; rcStatus('החיבור עבר לחלון אחר של האפליקציה.'); return; }
  if(m.t==='err'){ rcPending=null; const t={password:'הסיסמה לא נכונה',locked:'יותר מדי ניסיונות. נסה שוב בעוד 10 דקות.',badcode:'הקוד לא נכון או שפג תוקפו'}[m.code]||'שגיאה'; toast(t); rcStatus(t); if(m.code==='password'&&RCS.role==='host'&&!RCS.token){ RCS.role=null; rcSave(); } return; }
  if(m.t==='msg') rcMsg(m.d);
}
async function showPair(code){
  // on the computer's own server (localhost) the phone needs the computer's address on the home WiFi
  let base=location.origin;
  if(/^(localhost|127\.|\[::1\])/.test(location.hostname)){ try{ const j=await (await fetch('/lan')).json(); if(j.url) base=j.url; }catch(e){} }
  const url=`${base}${location.pathname}?pair=${code}`;
  $('remPairBox').hidden=false; $('remCode').textContent=code.replace(/(\d{3})(\d{3})/,'$1 $2'); $('remUrl').textContent=url;
  try{ const q=qrcode(0,'M'); q.addData(url); q.make(); $('remQR').innerHTML=q.createSvgTag({cellSize:5,margin:2,scalable:true}); }catch(e){ $('remQR').textContent=''; }
}

// a bar is an array with extra fields (seed, accents, no-extension); JSON drops those from arrays, so bars travel as history entries
function deser(st){ const bars=st.bars.map(b=>{ if(b.b) return histBar(b); const a=b.g; a.seed=b.s; return a; }); return {bars,num:st.num,den:st.den}; }
function rcMsg(d){
  if(RCS.role==='remote'){
    if(d.k==='state'){ RC.applying=true; const tabCh=d.S.tab!==S.tab; Object.assign(S,d.S); if(tabCh) setTab(S.tab); syncUI(); rhythm=deser(d); render(); RC.applying=false; if(RC.live) applyLive(RC.live); }
    if(d.k==='live') applyLive(d);
    if(d.k==='note') toast(d.txt,{ms:5000});
    return;
  }
  if(RCS.role==='host'){
    if(d.k==='S') applyPhoneS(d.S);
    if(d.k==='cmd'){ if(d.c==='toggle'&&!audOk()){ audGate(true); rsend({k:'note',txt:'צריך ללחוץ פעם אחת על מסך המחשב כדי לאפשר צליל, ואז ללחוץ נגן שוב'}); return; } if(d.c==='toggle') toggle(); if(d.c==='new') requestNew(); if(d.c==='clock'){ if(d.a==='go') clk.on?clockStop():clockStart(); if(d.a==='reset'){ clk.acc=0; if(clk.on) clk.from=performance.now(); clockTick(); } } hostSendLive(true); }
  }
}

  syncUI();
  const ch=k=>JSON.stringify(old[k])!==JSON.stringify(S[k]);
  if(old.bpm!==S.bpm) retime(old.bpm);
  if(['grids','level','num','den','bars'].some(ch)) requestNew(); else if(['restLvl','ties','showCount','sextCount','stick','dynOn'].some(ch)&&rhythm) render();
  RC.applying=false; save(); hostSendState(true);
}

async function rcLocalInit(){
  let j=null; try{ j=await (await fetch('/lan',{cache:'no-store'})).json(); }catch(e){ return false; } if(!j||!j.local) return false; rcLocal=j;
  const loop=/^(localhost|127\.0\.0\.1|\[::1\])$/.test(location.hostname);
  $('remEnterCode').hidden=true; $('remPair').hidden=true;
  if(loop){
    RCS.role='host'; rcSave(); rcStopped=false; rcConnect();
    const url=(j.url||location.origin)+'/';
    $('remPairBox').hidden=false; $('remCode').hidden=true; $('remUrl').textContent=url;
    const h=$('remPairBox').querySelector('.hint'); if(h) h.textContent='הטלפון צריך להיות על אותו WiFi. אפשר לשמור את הכתובת למסך הבית.';
    try{ const q=qrcode(0,'M'); q.addData(url); q.make(); $('remQR').innerHTML=q.createSvgTag({cellSize:5,margin:2,scalable:true}); }catch(e){}
  } else {
    if(!(RCS.role==='remote'&&RCS.token)) rcPending={auto:true};
    rcStopped=false; rcConnect();
  }
  rcUI(); return true;
}
