/* ---------- rhythm generation ---------- */
const density=()=>0.15+(S.level-1)*0.8/9;
function groupsFor(num,den){
  if(den===4) return Array(num).fill(1);
  if(num%3===0) return Array(num/3).fill(3);
  if(num===1) return [1];
  if(num%2===0) return Array(num/2).fill(2);
  return Array((num-3)/2).fill(2).concat(3);
}
// difficulty score of one pattern (same formula as the rhythm map)
function diffRaw(code){
  const [fam,bits]=code.split(':');
  if(fam==='Q'||fam==='H'){ let r=fam==='Q'?3:4; if(bits[0]!=='1') r+=2;
    for(let c=0;c<3;c++){ const cell=bits.slice(c*2,c*2+2); if(cell==='01') r+=1.5; if(cell==='00'&&bits.slice(0,c*2).includes('1')&&bits.slice(c*2+2).includes('1')) r+=.5; }
    return r; }
  const g=+fam, b=[...bits].map(x=>x==='1');
  if(!b.some(Boolean)) return 0;
  let r={4:1,3:2,6:3}[g];
  if(g===4&&b.every((x,i)=>!x||i%2===0)) r=0;
  if(!b[0]) r+=2;
  let strong=g===4?[0,2]:g===3?[0]:[0,3];
  if(g===6&&bits.indexOf('1')%2===0) strong=[0,2,3,4];
  b.forEach((x,i)=>{ if(x&&!strong.includes(i)&&!b[i-1]) r+=1.5; });
  const first=b.indexOf(true), last=b.lastIndexOf(true);
  for(let i=first+1;i<last;i++) if(!b[i]&&b[i-1]) r+=.5;
  if(g===6){ const okH=h=>h==='100'||h==='111'||h==='000', dOk=okH(bits.slice(0,3))&&okH(bits.slice(3)), tOk=bits.indexOf('1')%2===0; if(!dOk&&!tOk) r+=3; else if(!dOk) r+=1; }
  return r;
}
const DIFF_EDIT={'6:110010':10,'6:110100':10,'6:110110':9,'6:111000':7,'6:111010':8};
const diffOf=code=>DIFF_EDIT[code]!=null?DIFF_EDIT[code]:Math.max(1,Math.min(10,Math.round(1+diffRaw(code))));
const POOLS={};
// every distinct one-beat pattern the selected subdivisions can make, written in the simplest selected grid
function qPool(){
  let sel=[1,2,4,3,6].filter(g=>S.grids[g]); if(!sel.length) sel=S.grids.L?[3]:[2];
  const key='q'+sel.join(); if(POOLS[key]) return POOLS[key];
  const m=new Map();
  sel.forEach(g=>{ for(let x=0;x<(1<<g);x++){ const bt={t:'q',k:1,g,slots:[...Array(g)].map((_,i)=>!!((x>>(g-1-i))&1))}; const c=beatCode(bt); if(EXCL.has(c)||m.has(c)) continue; m.set(c,{bt,d:diffOf(c),rest:!bt.slots.some(Boolean)}); } });
  return POOLS[key]=[...m.values()];
}
function longPool(k){
  const key='L'+k; if(POOLS[key]) return POOLS[key];
  const out=[];
  ['10','11','00'].forEach(a=>['10','11','00'].forEach(b=>['10','11','00'].forEach(c=>{ const x=a+b+c; if(x==='000000') return; const bt={t:'L',k,g:6,slots:[...x].map(ch=>ch==='1')}; const code=beatCode(bt); if(!EXCL.has(code)) out.push({bt,d:diffOf(code)}); })));
  return POOLS[key]=out;
}
// complexity 1..10 stretched over the easiest..hardest pattern the current choice allows
function target(){
  let all=qPool(); if(S.grids.L&&S.den===4) all=all.concat(longPool(2),longPool(4));
  const ds=all.map(p=>p.d), lo=Math.min(...ds), hi=Math.max(...ds);
  return lo+(S.level-1)/9*(hi-lo);
}
function pick(pool,T){
  const w=pool.map(p=>p.d<=T+1?Math.exp(-Math.abs(T-p.d)/1.5)*(p.rest?0.3:1):0);
  let sum=w.reduce((a,b)=>a+b,0);
  if(sum<=0){ const lo=Math.min(...pool.map(p=>p.d)); const easy=pool.filter(p=>p.d===lo); const p=easy[Math.floor(Math.random()*easy.length)]; return {...p.bt,slots:p.bt.slots.slice()}; }
  let r=Math.random()*sum;
  for(let i=0;i<pool.length;i++){ r-=w[i]; if(r<=0) return {...pool[i].bt,slots:pool[i].bt.slots.slice()}; }
  const p=pool[pool.length-1]; return {...p.bt,slots:p.bt.slots.slice()};
}
function genGroup(k,den,T){
  if(den===4) return pick(qPool(),T);
  const d=density();
  const opts=[]; if(S.grids[1]||S.grids[2]||S.grids[3]||S.grids.L) opts.push(k); if(S.grids[4]||S.grids[6]) opts.push(k*2); if(!opts.length) opts.push(k);
  const g=opts[Math.floor(Math.random()*opts.length)];
  return {t:'e',k,g,slots:Array.from({length:g},()=>Math.random()<d)};
}
function genLong(k,T){ const p=longPool(k); return p.length?pick(p,T):null; }
// variety: at higher levels only some beats aim at the top difficulty, the rest are drawn from lower down,
// no pattern repeats inside a bar, both duple and triple feels show up when both are chosen, and a full beat appears now and then as an anchor
const famOf=bt=>bt.t==='L'||bt.g===3||bt.g===6?'t':'b';
function beatT(T,lo){
  if(S.level<4||Math.random()<0.55) return T;
  return lo+(T-lo)*(0.2+0.65*Math.random());
}
function pickVar(T,lo,used,fam){
  let pl=qPool();
  if(fam){ const f=pl.filter(p=>famOf(p.bt)===fam); if(f.length) pl=f; }
  if(S.level>=6&&Math.random()<0.12){ const full=pl.filter(p=>p.bt.g>1&&p.bt.slots.every(Boolean)&&!used.has(beatCode(p.bt))); if(full.length){ const p=full[Math.floor(Math.random()*full.length)]; return {...p.bt,slots:p.bt.slots.slice()}; } }
  const fresh=pl.filter(p=>!used.has(beatCode(p.bt))&&p.d<=T+1); if(fresh.length) pl=fresh;
  return pick(pl,beatT(T,lo));
}
/* ---------- path mode: a teaching ladder; each level adds patterns, earlier ones stay for review ---------- */
const LADDER_D={'4:1000':1,'4:1010':1,'4:0000':2,'4:0010':2,'4:1111':3,'4:1011':5,'4:1110':5,'4:0111':5,'4:0011':5,'4:1001':6,'4:1100':6,'4:1101':7,'4:0100':8,'4:0110':8,'4:0001':8,'4:0101':9};
const LADDER_T={'3:111':1,'3:101':2,'3:110':3,'3:011':3,'Q:101010':4,'3:010':5,'3:001':5};
const DESC_D=['רבע, 2 שמיניות','הפסקות: רבע, שמינית + שמינית','4 שש־עשריות','רבע עם נקודה בין פעמות','שמינית + 2 שש־עשריות ולהפך, הפסקה + 3 שש־עשריות','שמינית עם נקודה + שש־עשרית ולהפך','שש־עשרית־שמינית־שש־עשרית','סינקופות על ה-e וה-a','הפסקות פנימיות קשות','הכל, כולל שילובים צפופים'];
const DESC_T=['טריולה מלאה','טריולה עם הפסקה באמצע','1 & · ו־· & a','טריולת רבעים מלאה','תו בודד על ה-& או על ה-a','טריולת רבעים עם הפסקות, טריולת חצאים','סקסטולה מלאה וסקסטולות פשוטות','סקסטולות בינוניות','סקסטולות מעורבות','הכל'];
// level fixes sent from the rhythm map (code: level, 0 = not in the path)
const LVL_EDIT={};
let ALLP=null;
// every pattern the app can write, once: one-beat patterns in their simplest grid plus the long triplets
function allPatterns(){
  if(ALLP) return ALLP;
  const m=new Map();
  [1,2,4,3,6].forEach(g=>{ for(let x=0;x<(1<<g);x++){ const bt={t:'q',k:1,g,slots:[...Array(g)].map((_,i)=>!!((x>>(g-1-i))&1))}; const c=beatCode(bt); if(!m.has(c)) m.set(c,{bt,code:c,k:1}); } });
  [2,4].forEach(k=>['10','11','00'].forEach(a=>['10','11','00'].forEach(b=>['10','11','00'].forEach(c=>{ const x=a+b+c; if(x==='000000') return; const bt={t:'L',k,g:6,slots:[...x].map(ch=>ch==='1')}; const code=beatCode(bt); if(!m.has(code)) m.set(code,{bt,code,k}); }))));
  const list=[...m.values()];
  // sextuplets that stay in the app are split into three steps by difficulty
  const sx=list.filter(p=>p.code[0]==='6'&&!EXCL.has(p.code)).sort((a,b)=>diffOf(a.code)-diffOf(b.code));
  const sxL={}; sx.forEach((p,i)=>{ sxL[p.code]=p.code==='6:111111'?7:7+Math.min(2,Math.floor(3*i/sx.length)); });
  list.forEach(p=>{
    const c=p.code; p.trk=c[0]==='4'?'d':'t';
    p.lvl=LVL_EDIT[c]!=null?LVL_EDIT[c]:LADDER_D[c]||LADDER_T[c]||sxL[c]||((c[0]==='Q'||c[0]==='H')&&!EXCL.has(c)?6:0);
    p.ok=c==='Q:101010'||!EXCL.has(c)||!!LVL_EDIT[c];
  });
  return ALLP=list;
}
const modeOf=()=>S.mode||'free';
// weighted pool for path and manual modes; in path the newest level gets about 40% of the beats
function poolNow(){
  const all=allPatterns();
  if(modeOf()==='manual'){ const on=new Set(S.manual||[]); let pl=all.filter(p=>on.has(p.code)); if(!pl.length) pl=all.filter(p=>p.code==='4:1000'); return pl.map(p=>({p,w:1})); }
  const L=S.level, dOn=S.trkD!==false, tOn=!!S.trkT||!dOn;
  const pl=all.filter(p=>p.ok&&p.lvl&&p.lvl<=L&&(p.trk==='d'?dOn:tOn));
  const nw=pl.filter(p=>p.lvl===L).length, od=pl.length-nw;
  return pl.map(p=>{ let w=nw&&od?(p.lvl===L?0.4/nw:0.6/od):1/pl.length; if(p.code==='4:0000') w*=0.5; if(p.k>1) w*=0.25; return {p,w}; });
}
function drawW(items){ let s=items.reduce((a,b)=>a+b.w,0), r=Math.random()*s; for(const it of items){ r-=it.w; if(r<=0) return it.p; } return items[items.length-1].p; }
function genLadderBar(){
  const pool=poolNow(), q=pool.filter(x=>x.p.k===1), used=new Set(), gr=[]; let i=0;
  while(i<S.num){
    const fit=pool.filter(x=>x.p.k===1||(i%2===0&&i+x.p.k<=S.num));
    let p=drawW(fit.length?fit:pool);
    if(p.k>1&&(i%2||i+p.k>S.num)) p=null;
    if(p&&p.k===1&&used.has(p.code)){ const fresh=q.filter(x=>!used.has(x.p.code)); if(fresh.length) p=drawW(fresh); }
    if(!p){ p=drawW(q.length?q:pool.filter(x=>x.p.k===1)); }
    if(!p){ gr.push({t:'q',k:1,g:1,slots:[true]}); i++; continue; }
    used.add(p.code); gr.push(Object.assign({},p.bt,{slots:p.bt.slots.slice()})); i+=p.k;
  }
  if(modeOf()==='path'&&S.level<4) gr.noExt=true;
  return gr;
}
function genRhythm(){
  const groups=groupsFor(S.num,S.den), bars=[], T=target();
  const ds=qPool().map(p=>p.d), lo=Math.min(...ds);
  const only=![1,2,4,3,6].some(g=>S.grids[g]);
  const hasB=S.grids[2]||S.grids[4], hasT=S.grids[3]||S.grids[6]||S.grids.L;
  let lastCode='';
  for(let b=0;b<S.bars;b++){
    let gr=[];
    if(S.den===4&&modeOf()!=='free'){ gr=genLadderBar(); }
    else if(S.den===4){
      const used=new Set(); if(lastCode) used.add(lastCode);
      let i=0;
      while(i<S.num){
        if(S.grids.L){
          const r=Math.random(), fitsL=k=>longPool(k).some(p=>p.d<=T+1), h=fitsL(4), qq=fitsL(2);
          if(h&&i%2===0&&i+4<=S.num&&r<(only?(qq?0.35:1):0.12)){ gr.push(genLong(4,T)); i+=4; continue; }
          if(qq&&i%2===0&&i+2<=S.num&&r<(only?1:0.35)){ gr.push(genLong(2,T)); i+=2; continue; }
        }
        const bt=pickVar(T,lo,used); used.add(beatCode(bt)); gr.push(bt); i++;
      }
      // both feels chosen: from level 5 a bar holds at least one of each
      if(S.level>=5&&hasB&&hasT&&!only&&gr.length>=2){
        const fams=new Set(gr.map(famOf));
        if(fams.size===1){ const need=fams.has('b')?'t':'b', qs=gr.map((x,j)=>x.t==='q'?j:-1).filter(j=>j>0);
          if(qs.length){ const j=qs[Math.floor(Math.random()*qs.length)], nb=pickVar(T,lo,used,need); if(famOf(nb)===need) gr[j]=nb; } }
      }
      const lq=gr[gr.length-1]; lastCode=lq&&lq.t==='q'?beatCode(lq):'';
    } else gr=groups.map(k=>genGroup(k,S.den,T));
    if(!gr.some(x=>x.slots.some(Boolean))){ const x=gr[Math.floor(Math.random()*gr.length)]; x.slots[0]=true; }
    gr.seed=1+Math.floor(Math.random()*2e9);
    if(S.dynOn) gr.acc=pickAccents(gr,S.den,dynStage());
    bars.push(gr);
  }
  return {bars,num:S.num,den:S.den};
}

/* ---------- dynamics: accents on some notes, played with singles by the smallest value in each beat ---------- */
const DYN_N=['רק יד ימין','רק יד שמאל','שתי הידיים, פשוט','יותר אקצנטים בתיבה','אקצנטים צמודים','על המקומות החלשים','הכל מעורבב'];
const dynStage=()=>S.dynAuto!==false?[1,1,2,3,4,5,5,6,6,7][clamp(S.level,1,10)-1]:clamp(S.dynLvl||1,1,7);
// onset tick -> 'R' / 'L'. Singles by the smallest value in each beat; an odd count (triplet) carries the next hand into the next beat; quarters are always right
function stickMap(bar,den){
  const TPU=den===8?6:12, m=new Map(); let off=0, u=0;
  bar.forEach(bt=>{
    const on=[]; bt.slots.forEach((h,i)=>{ if(h) on.push(i); });
    if(bt.t==='L'){ const has11=[0,1,2].some(c=>bt.slots[c*2]&&bt.slots[c*2+1]), G=has11?6:3;
      for(let i=0;i<G;i++){ const si=i*6/G; if(bt.slots[si]) m.set(Math.round((u+si*bt.k/6)*TPU),(off+i)%2?'L':'R'); }
      if(G%2) off=(off+G)%2; u+=bt.k; return; }
    const g=bt.g, G=[...Array(g)].map((_,i)=>i+1).find(d=>g%d===0&&on.every(i=>Number.isInteger(i*d/g)))||g;
    if(on.length){
      if(G===1) m.set(Math.round(u*TPU),'R');
      else { for(let j=0;j<G;j++){ const si=j*g/G; if(bt.slots[si]) m.set(Math.round((u+si*bt.k/g)*TPU),(off+j)%2?'L':'R'); } if(G%2) off=(off+G)%2; }
    }
    u+=bt.k;
  });
  return m;
}
function pickAccents(bar,den,stage){
  const TPU=den===8?6:12, hm=stickMap(bar,den), ons=[...hm.keys()].sort((a,b)=>a-b);
  if(!ons.length) return [];
  const pos=t=>t%TPU, simple=t=>pos(t)===0||pos(t)*2===TPU, weak=t=>!simple(t);
  const R=(a,b)=>a+Math.floor(Math.random()*(b-a+1));
  const shuffle=a=>{ for(let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; } return a; };
  let cand=ons, n=R(1,2);
  if(stage===1) cand=ons.filter(t=>hm.get(t)==='R');
  if(stage===2) cand=ons.filter(t=>hm.get(t)==='L');
  if(stage===3){ cand=ons.filter(simple); }
  if(stage===4||stage===6) n=R(2,4);
  if(stage===5) n=R(3,5);
  if(stage===7) n=R(3,6);
  if(!cand.length) cand=ons;
  const out=new Set(), MX=clamp(S.dynMax||3,1,8), idx=new Map(ons.map((t,i)=>[t,i]));
  // no more than MX accented notes in a row
  const fits=t=>{ const i=idx.get(t); let run=1; for(let j=i-1;j>=0&&out.has(ons[j]);j--) run++; for(let j=i+1;j<ons.length&&out.has(ons[j]);j++) run++; return run<=MX; };
  if(stage===5&&MX>=2){ const pairs=ons.slice(1).map((t,i)=>[ons[i],t]).filter(([a,b])=>hm.get(a)!==hm.get(b)); if(pairs.length){ const p=pairs[Math.floor(Math.random()*pairs.length)]; out.add(p[0]); out.add(p[1]); } }
  if(stage===6){ const w=ons.filter(weak); if(w.length) out.add(w[Math.floor(Math.random()*w.length)]); }
  shuffle(cand.slice()).forEach(t=>{ if(out.size<Math.min(n,cand.length+out.size)&&fits(t)) out.add(t); });
  return [...out].sort((a,b)=>a-b);
}
