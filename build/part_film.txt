/* ---------- tap play-along: tap the pad with the notes, get scored ---------- */
var TAP={exp:[],bars:[],good:0,ok:0,bad:0,miss:0,extra:0,offs:[],streak:0,best:0};
var FILM={on:false,run:false,round:0,rounds:1,orig:null,stream:null,rec:null,chunks:[],recOn:false,ending:false,t0:0,raf:0,blob:null,url:null,cv:null,dest:null,destOn:false,micSrc:null,discard:false,ptrs:new Set(),lp:0,lastDraw:0};
const tapLat=()=>ctx?(ctx.outputLatency||0)+(ctx.baseLatency||0):0;
// event time on the audio clock, with the device's output delay taken off
const tapTime=ev=>ctx.currentTime-Math.max(0,(performance.now()-ev.timeStamp)/1000)-tapLat();
function tapOnNow(){ return !!S.tapOn&&S.tab==='gen'&&!(FILM&&FILM.on)&&!RC.remoteMode; }
function tapUI(){ const on=!!S.tapOn&&S.tab==='gen'; $('pad').hidden=!on; $('tapStats').hidden=!on; tapShow(); requestAnimationFrame(fitScore); }
function tapReset(){ Object.assign(TAP,{exp:[],bars:[],good:0,ok:0,bad:0,miss:0,extra:0,offs:[],streak:0,best:0}); tapShow(); }
function tapExpect(t,id,k,bs,be){ TAP.exp.push({t,id,bk:k,done:false}); if(!TAP.bars.some(b=>b.k===k)) TAP.bars.push({k,start:bs,end:be,bad:false}); }
function tapMark(id,c){ const el=$(id); if(el){ el.classList.remove('tg','ty','tx'); el.classList.add(c); } }
function tapBad(k){ const b=TAP.bars.find(x=>x.k===k); if(b) b.bad=true; }
function tapHit(ev){
  if(!playing||!ctx||!tapOnNow()) return null;
  const t=tapTime(ev)-(S.tapCal||0)/1000;
  let best=null, bd=0.15;
  TAP.exp.forEach(e=>{ if(e.done) return; const d=Math.abs(t-e.t); if(d<bd){ bd=d; best=e; } });
  let cls;
  if(best){
    best.done=true; const d=t-best.t, a=Math.abs(d); TAP.offs.push(d);
    if(a<=0.04){ TAP.good++; cls='tg'; } else if(a<=0.09){ TAP.ok++; cls='ty'; } else { TAP.bad++; cls='tx'; tapBad(best.bk); }
    tapMark(best.id,cls);
  } else {
    const b=TAP.bars.find(x=>t>=x.start-0.05&&t<=x.end); if(!b) return 'tg';
    TAP.extra++; b.bad=true; cls='tx';
  }
  tapShow(); return cls;
}
function tapSweep(){
  const now=ctx.currentTime-tapLat()-(S.tapCal||0)/1000; let ch=false;
  TAP.exp=TAP.exp.filter(e=>{ if(e.done) return false; if(now>e.t+0.15){ TAP.miss++; tapMark(e.id,'tx'); tapBad(e.bk); ch=true; return false; } return true; });
  TAP.bars=TAP.bars.filter(b=>{ if(now>b.end+0.16){ if(b.bad) TAP.streak=0; else { TAP.streak++; TAP.best=Math.max(TAP.best,TAP.streak); } ch=true; return false; } return true; });
  if(ch) tapShow();
}
function tapNums(){
  const tot=TAP.good+TAP.ok+TAP.bad+TAP.miss+TAP.extra, acc=tot?Math.round(100*(TAP.good+0.6*TAP.ok+0.2*TAP.bad)/tot):null;
  const m=TAP.offs.length?TAP.offs.reduce((a,b)=>a+b,0)/TAP.offs.length*1000:0;
  return {tot,acc,m,has:TAP.offs.length>0};
}
function tapShow(){
  if(!$('tapStats')||$('tapStats').hidden) return;
  const {acc,m,has}=tapNums(), mid=!has||Math.abs(m)<8;
  $('tAcc').textContent=acc==null?'—':acc+'%';
  // tendency as a dot on a small meter: left = early, right = late
  const tm=$('tTend'); tm.classList.toggle('none',!has); tm.firstChild.style.left=(has?50+clamp(m/90,-1,1)*44:50)+'%';
  $('tOffL').textContent=!has?'נטייה':mid?'בזמן':m<0?'מקדים':'מאחר';
  $('tStreak').textContent=TAP.streak; $('tBest').textContent=TAP.best;
}
function tapSummary(){
  const {tot,acc,m,has}=tapNums(); if(!tot) return;
  const lean=!has||Math.abs(m)<8?'בזמן':m<0?'נוטה להקדים':'נוטה לאחר';
  toast(`${acc}% דיוק · ${lean} · שיא רצף ${TAP.best}`);
}

/* recording, all on one clock:
   1. the screen (notes) is painted onto a 1080x1920 canvas and recorded as video only
   2. camera + microphone go into one file, so the two stay in sync by themselves
   3. the app's sound is not recorded at all: every sound played is logged, and rebuilt clean when saving
   a small tap on the audio clock (app sound left, mic right) tells the editor exactly where the camera file sits on that clock */
const REC_TYPES=['video/mp4;codecs=avc1.42E01E,mp4a.40.2','video/mp4','video/webm;codecs=vp9,opus','video/webm;codecs=vp8,opus','video/webm'];

const AUD_TYPES=['audio/mp4','audio/webm;codecs=opus','audio/webm','audio/ogg;codecs=opus'];
const pickType=L=>L.find(t=>MediaRecorder.isTypeSupported&&MediaRecorder.isTypeSupported(t))||'';

// loudness of the app sound and the mic, one value per millisecond, on the audio clock
function recCapStart(stream){
  const C={t0:null,n:0,sr:ctx.sampleRate,L:[],R:[],mic:false};
  try{
    const merge=ctx.createChannelMerger(2), sp=ctx.createScriptProcessor(2048,2,2), mute=ctx.createGain(); mute.gain.value=0;
    master.connect(merge,0,0);
    if(stream&&stream.getAudioTracks().length){ C.src=ctx.createMediaStreamSource(new MediaStream(stream.getAudioTracks())); C.src.connect(merge,0,1); C.mic=true; }
    merge.connect(sp); sp.connect(mute); mute.connect(ctx.destination);
    sp.onaudioprocess=e=>{ if(!REC.on&&C.t0!=null) return; if(C.t0==null) C.t0=e.playbackTime-REC.t0;
      const a=e.inputBuffer.getChannelData(0), b=e.inputBuffer.getChannelData(1), L=C.L, R=C.R, f=1000/C.sr;
      for(let i=0;i<a.length;i++){ const k=Math.floor((C.n+i)*f), x=a[i]<0?-a[i]:a[i], y=b[i]<0?-b[i]:b[i];
        if(L[k]===undefined){ L[k]=x; R[k]=y; } else { if(x>L[k]) L[k]=x; if(y>R[k]) R[k]=y; } }
      C.n+=a.length; };
    C.stop=()=>{ try{ master.disconnect(merge); }catch(e){} try{ C.src&&C.src.disconnect(); merge.disconnect(); sp.disconnect(); mute.disconnect(); }catch(e){} sp.onaudioprocess=null; };
  }catch(e){ C.stop=()=>{}; }
  return C;
}

// paint the live frame (boxes, text and the notation SVG) onto the canvas
let FR=null;
const visCol=c=>!!c&&c!=='transparent'&&c!=='none'&&!/rgba\([^)]*,\s*0\)$/.test(c);
function rrP(p,x,y,w,h,r){ r=Math.max(0,Math.min(r,w/2,h/2)); p.moveTo(x+r,y); p.arcTo(x+w,y,x+w,y+h,r); p.arcTo(x+w,y+h,x,y+h,r); p.arcTo(x,y+h,x,y,r); p.arcTo(x,y,x+w,y,r); p.closePath(); }
function filmDraw(){
  const cv=FILM.cv, g=cv.getContext('2d'), fr=$('fFrame').getBoundingClientRect(); if(!fr.width) return;
  FR={g,ox:fr.left,oy:fr.top,sc:cv.width/fr.width};
  g.setTransform(1,0,0,1,0,0); g.globalAlpha=1; g.fillStyle=getComputedStyle($('fFrame')).backgroundColor; g.fillRect(0,0,cv.width,cv.height);
  drawNode($('fFrame'));
}
const baseT=()=>FR.g.setTransform(FR.sc,0,0,FR.sc,-FR.ox*FR.sc,-FR.oy*FR.sc);
function drawNode(el){
  if(el.hasAttribute('data-norec')) return;
  const cs=getComputedStyle(el); if(cs.display==='none'||cs.visibility==='hidden') return;
  if(el.id==='fCam'){ drawCam(el,cs); return; }
  if(el.tagName.toLowerCase()==='svg'){ drawSvg(el); return; }
  paintBox(el,cs);
  for(const ch of el.childNodes){ if(ch.nodeType===1) drawNode(ch); else if(ch.nodeType===3) drawText(ch,cs); }
}
function paintBox(el,cs){
  const bg=cs.backgroundColor, bw=parseFloat(cs.borderTopWidth)||0, bc=cs.borderTopColor, hasB=bw>0&&cs.borderTopStyle!=='none'&&visCol(bc);
  if(!visCol(bg)&&!hasB) return;
  const g=FR.g, r=el.getBoundingClientRect(); if(!r.width||!r.height) return;
  let rad=cs.borderTopLeftRadius; rad=rad.endsWith('%')?Math.min(r.width,r.height)*parseFloat(rad)/100:(parseFloat(rad)||0);
  baseT(); g.globalAlpha=+cs.opacity;
  if(visCol(bg)){ g.fillStyle=bg; g.beginPath(); rrP(g,r.left,r.top,r.width,r.height,rad); g.fill(); }
  if(hasB){ g.strokeStyle=bc; g.lineWidth=bw; g.beginPath(); rrP(g,r.left+bw/2,r.top+bw/2,r.width-bw,r.height-bw,Math.max(0,rad-bw/2)); g.stroke(); }
  g.globalAlpha=1;
}
const txtRange=document.createRange();
function drawText(node,cs){
  const s=node.textContent; if(!s.trim()) return;
  const g=FR.g, re=/\S+/g; let m;
  baseT(); g.globalAlpha=+cs.opacity;
  g.font=`${cs.fontStyle} ${cs.fontWeight} ${cs.fontSize} ${cs.fontFamily}`; g.fillStyle=cs.color; g.textAlign='center'; g.textBaseline='middle'; g.direction=cs.direction;
  if(cs.textShadow&&cs.textShadow!=='none'){ g.shadowColor='rgba(0,0,0,.8)'; g.shadowBlur=12*FR.sc; }
  while((m=re.exec(s))){ txtRange.setStart(node,m.index); txtRange.setEnd(node,m.index+m[0].length); const r=txtRange.getBoundingClientRect(); if(!r.width) continue; g.fillText(m[0],r.left+r.width/2,r.top+r.height/2); }
  g.shadowBlur=0; g.shadowColor='transparent'; g.globalAlpha=1;
}
function drawSvg(svg){
  const {g,ox,oy,sc}=FR;
  svg.querySelectorAll('line,rect,path,ellipse,text').forEach(el=>{
    const cs=getComputedStyle(el); if(cs.display==='none'||cs.visibility==='hidden') return;
    const m=el.getScreenCTM(); if(!m) return;
    g.setTransform(sc*m.a,sc*m.b,sc*m.c,sc*m.d,sc*(m.e-ox),sc*(m.f-oy));
    g.globalAlpha=(+cs.opacity)*(cs.fillOpacity!==''?+cs.fillOpacity:1);
    const fill=cs.fill, stroke=cs.stroke, sw=parseFloat(cs.strokeWidth)||0, tag=el.tagName.toLowerCase();
    const doFill=visCol(fill)&&tag!=='line', doStroke=visCol(stroke)&&sw>0;
    if(cs.filter&&cs.filter!=='none'&&doFill){ g.shadowColor=fill; g.shadowBlur=6*sc; }
    if(tag==='text'){
      const an=cs.textAnchor; g.font=`${cs.fontStyle} ${cs.fontWeight} ${cs.fontSize} ${cs.fontFamily}`;
      g.textAlign=an==='middle'?'center':an==='end'?'right':'left'; g.textBaseline='alphabetic'; g.direction='ltr';
      const x=el.x.baseVal.length?el.x.baseVal.getItem(0).value:0, y=el.y.baseVal.length?el.y.baseVal.getItem(0).value:0, t=el.textContent;
      if(doStroke){ g.lineWidth=sw; g.strokeStyle=stroke; g.lineJoin='round'; g.strokeText(t,x,y); }
      if(doFill){ g.fillStyle=fill; g.fillText(t,x,y); }
    } else {
      const p=new Path2D();
      if(tag==='line'){ p.moveTo(el.x1.baseVal.value,el.y1.baseVal.value); p.lineTo(el.x2.baseVal.value,el.y2.baseVal.value); }
      else if(tag==='rect'){ const x=el.x.baseVal.value,y=el.y.baseVal.value,w=el.width.baseVal.value,h=el.height.baseVal.value,r=el.rx.baseVal.value||el.ry.baseVal.value||0; if(r) rrP(p,x,y,w,h,r); else p.rect(x,y,w,h); }
      else if(tag==='ellipse'){ p.ellipse(el.cx.baseVal.value,el.cy.baseVal.value,el.rx.baseVal.value,el.ry.baseVal.value,0,0,Math.PI*2); }
      else p.addPath(new Path2D(el.getAttribute('d')||''));
      if(doFill){ g.fillStyle=fill; g.fill(p); }
      if(doStroke){ g.lineWidth=sw; g.strokeStyle=stroke; g.lineCap=cs.strokeLinecap||'butt'; g.stroke(p); }
    }
    g.shadowBlur=0; g.shadowColor='transparent'; g.globalAlpha=1;
  });
}
function drawCam(el,cs){
  const v=$('fVid'); if(!v.videoWidth) return;
  const g=FR.g, r=el.getBoundingClientRect(), rad=Math.min(r.width/2,/%$/.test(cs.borderTopLeftRadius)?r.width*parseFloat(cs.borderTopLeftRadius)/100:(parseFloat(cs.borderTopLeftRadius)||0)), bw=parseFloat(cs.borderTopWidth)||0;
  baseT(); g.save(); g.beginPath(); rrP(g,r.left,r.top,r.width,r.height,rad); g.clip();
  const s=Math.max(r.width/v.videoWidth,r.height/v.videoHeight), dw=v.videoWidth*s, dh=v.videoHeight*s;
  g.translate(r.left+r.width/2,r.top+r.height/2); g.scale(-1,1); g.drawImage(v,-dw/2,-dh/2,dw,dh); g.restore();
  if(bw){ baseT(); g.strokeStyle=cs.borderTopColor; g.lineWidth=bw; g.beginPath(); rrP(g,r.left+bw/2,r.top+bw/2,r.width-bw,r.height-bw,Math.max(0,rad-bw/2)); g.stroke(); }
}

/* ---------- headphone sync: 8 clicks at 60 BPM in eighths, hit along; the mic tells how late the sound reaches the ears ---------- */
let SY=null;
function syOpen(thenFilm){ fsCamStop(); $('fSyncScr').hidden=false; $('syIntro').hidden=false; $('syRun').hidden=true; $('syRes').hidden=true; SY={thenFilm}; }
function syClose(){ if(SY&&SY.cleanup) SY.cleanup(); SY=null; $('fSyncScr').hidden=true; if(!$('fSet').hidden) fsCam(); }
async function syGo(){
  await initAudio(); if(typeof applyVol==='function') applyVol();
  let mic; try{ mic=await navigator.mediaDevices.getUserMedia({audio:{echoCancellation:false,noiseSuppression:false,autoGainControl:false}}); }catch(e){ toast('צריך גישה למיקרופון כדי לסנכרן'); return; }
  $('syIntro').hidden=true; $('syRes').hidden=true; $('syRun').hidden=false;
  $('syDots').innerHTML='<i></i>'.repeat(8);
  // left = what the app plays, right = the microphone, captured sample by sample on the same clock
  const merge=ctx.createChannelMerger(2), src=ctx.createMediaStreamSource(mic), sp=ctx.createScriptProcessor(2048,2,2), mute=ctx.createGain(); mute.gain.value=0;
  master.connect(merge,0,0); src.connect(merge,0,1); merge.connect(sp); sp.connect(mute); mute.connect(ctx.destination);
  const L=[], R=[]; let t0=null;
  sp.onaudioprocess=e=>{ if(t0==null) t0=e.playbackTime; L.push(new Float32Array(e.inputBuffer.getChannelData(0))); R.push(new Float32Array(e.inputBuffer.getChannelData(1))); };
  SY.cleanup=()=>{ try{ master.disconnect(merge); }catch(e){} try{ src.disconnect(); sp.disconnect(); mute.disconnect(); }catch(e){} mic.getTracks().forEach(t=>t.stop()); };
  const start=ctx.currentTime+0.6, gap=0.5, clicks=[];
  for(let i=0;i<8;i++){ const t=start+i*gap; clicks.push(t); click(t,i===0?'acc':'beat');
    setTimeout(()=>{ if(!SY) return; const p=$('syPulse'); p.textContent=8-i; p.classList.add('beat'); setTimeout(()=>p.classList.remove('beat'),90); const d=$('syDots').children[i]; if(d) d.classList.add('on'); },Math.max(0,(t-ctx.currentTime)*1000)); }
  setTimeout(()=>{ if(!SY) return; const cl=SY.cleanup; SY.cleanup=null; cl(); syAnalyze(L,R,t0,clicks); },(start-ctx.currentTime+8*gap+0.4)*1000);
}
function syAnalyze(Lc,Rc,t0,clicks){
  const sr=ctx.sampleRate, cat=a=>{ const n=a.reduce((s,x)=>s+x.length,0), o=new Float32Array(n); let p=0; a.forEach(x=>{ o.set(x,p); p+=x.length; }); return o; };
  const L=cat(Lc), R=cat(Rc), ms=Math.floor(sr/1000);
  const env=x=>{ const n=Math.floor(x.length/ms), e=new Float32Array(n); for(let i=0;i<n;i++){ let m=0; for(let k=i*ms;k<(i+1)*ms;k++){ const v=Math.abs(x[k]); if(v>m) m=v; } e[i]=m; } return e; };
  const eL=env(L), eR=env(R);
  // where each click really sits in the captured audio (left), and the first hit after it in the mic (right)
  const onset=(e,from,to,th)=>{ for(let i=Math.max(4,from);i<Math.min(e.length,to);i++) if(e[i]-e[i-4]>th) return i; return -1; };
  const mxL=Math.max(...eL), mxR=Math.max(...eR), offs=[], rows=[];
  clicks.forEach(c=>{ const ci=Math.round((c-t0)*1000), l=onset(eL,ci-60,ci+300,mxL*0.3); // the capture runs a little behind the clock if(l<0) return;
    const h=onset(eR,l-60,l+420,mxR*0.25); if(h<0) return; offs.push((h-l)/1000); rows.push((h-l)/1000); });
  const o=offs.slice().sort((a,b)=>a-b);
  $('syRun').hidden=true; $('syRes').hidden=false;
  const bad=(msg)=>{ $('syIcon').textContent='!'; $('syIcon').classList.add('bad'); $('syHead').textContent='המדידה לא הצליחה'; $('syNum').textContent=''; $('syMsg').textContent=msg; $('syHits').innerHTML=''; $('syStat').textContent=''; $('syGo').textContent=SY&&SY.thenFilm?'המשך בלי סנכרון ›':'סגור'; };
  $('syIcon').classList.remove('bad'); $('syIcon').textContent='✓';
  if(o.length<5) return bad(`נקלטו רק ${o.length} מכות מתוך 8. הכה חזק יותר, קרוב לטלפון`);
  const med=o[Math.floor(o.length/2)], spread=o[o.length-2]-o[1];
  if(spread>0.09) return bad('המכות היו מפוזרות מדי. נסה שוב, בדיוק עם הקליק');
  if(med>0.35) return bad('יצא איחור גדול מדי. כנראה רעש ברקע. נסה שוב במקום שקט');
  S.hpLag=S.hpMeas=+Math.max(0,med).toFixed(3); save();
  $('syHead').textContent=med<0.03?'בלי איחור מורגש':'האוזניות מאחרות ב־'; $('syNum').textContent=Math.round(med*1000)+'ms';
  $('syMsg').innerHTML='מעכשיו הצליל בסרטון יוזז בהתאם.<br>נשמר גם לפעמים הבאות.';
  $('syHits').innerHTML='<div class="sy-mut sm" style="text-align:right">⬜ קליק  🟧 המכה שלך</div>'+rows.slice(0,4).map((d,i)=>`<div class="sy-row"><span>${i+1}</span><div class="ln"><i class="k" style="left:20%"></i><i class="h" style="left:${20+Math.min(75,d*1000/4)}%"></i></div></div>`).join('');
  $('syStat').textContent=`${o.length} מתוך 8 מכות נקלטו · יציב`; $('syGo').textContent=SY&&SY.thenFilm?'המשך לצילום ›':'סיום';
}
