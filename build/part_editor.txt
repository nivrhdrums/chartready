function edReady(v){ return new Promise(res=>{
  const done=()=>{ v.removeEventListener('timeupdate',fix); res(); };
  // recorded webm has no length; jumping far ahead makes the browser work it out
  const fix=()=>{ if(isFinite(v.duration)){ v.currentTime=0; v.removeEventListener('timeupdate',fix); setTimeout(res,60); } };
  if(v.readyState>=1){ if(isFinite(v.duration)) return res(); v.addEventListener('timeupdate',fix); v.currentTime=1e101; return; }
  v.addEventListener('loadedmetadata',()=>{ if(isFinite(v.duration)) done(); else { v.addEventListener('timeupdate',fix); v.currentTime=1e101; } },{once:true});
  setTimeout(res,2500); }); }
const relY=id=>{ const f=$('fFrame').getBoundingClientRect(), r=$(id).getBoundingClientRect(); return r.height?((r.top+r.height/2)-f.top)/f.height:null; };
/* loudness tools: one value per millisecond */
function envOf(buf){ const d=buf.getChannelData(0), sr=buf.sampleRate, n=Math.floor(d.length*1000/sr)+1, e=new Float32Array(n), f=1000/sr;
  for(let i=0;i<d.length;i++){ const k=Math.floor(i*f), v=d[i]<0?-d[i]:d[i]; if(v>e[k]) e[k]=v; } return e; }
// rises in loudness = hits
function onsetsOf(env){ const n=env.length, o=new Float32Array(n); for(let i=4;i<n;i++){ const d=(env[i]||0)-(env[i-4]||0); o[i]=d>0?d:0; } return o; }
// the shift (ms) that best puts hits of `on` on the given times (ms)
function lagToTimes(on,times,lo,hi){ const ts=times.slice(0,64); if(ts.length<2) return null; let best=-1, bl=null;
  for(let L=lo;L<=hi;L++){ let s=0; for(const t of ts){ const c=Math.round(t)+L; let m=0; for(let j=c-3;j<=c+3;j++) if(j>=0&&j<on.length&&on[j]>m) m=on[j]; s+=m; } if(s>best){ best=s; bl=L; } }
  return best>0?bl:null; }
// the shift D (ms) with b[t+D] ≈ a[t]; coarse in 10ms steps, then exact. null when the match is weak
function matchShift(a,b,range){
  const pool=(x,k)=>{ const o=new Float32Array(Math.ceil(x.length/k)); for(let i=0;i<x.length;i++){ const j=(i/k)|0; if(x[i]>o[j]) o[j]=x[i]; } return o; };
  const score=(A,B,D,st)=>{ let s=0; for(let t=0;t<A.length;t+=st){ const u=t+D; if(u>=0&&u<B.length) s+=A[t]*B[u]; } return s; };
  const A10=pool(a,10), B10=pool(b,10), R=Math.round(range/10), sc=[]; let best=-1, bd=0;
  for(let D=-R;D<=R;D++){ const s=score(A10,B10,D,1); sc.push(s); if(s>best){ best=s; bd=D; } }
  const med=sc.slice().sort((x,y)=>x-y)[sc.length>>1]||0; if(!(best>0)||best<med*2.5) return null;
  let b1=-1, D1=bd*10; for(let D=bd*10-15;D<=bd*10+15;D++){ const s=score(a,b,D,1); if(s>b1){ b1=s; D1=D; } }
  return D1; }
/* the app sound, rebuilt from the log on a silent audio context: clicks and notes come out as two clean tracks */
async function edStem(log,dur,which,accR){
  const sr=ctx.sampleRate, off=new OfflineAudioContext(1,Math.max(1,Math.ceil((dur+1)*sr)),sr);
  const sv={ctx,master,OUT,bus:Object.assign({},BUS),snd:S.clickSnd}, out=off.createGain(); out.gain.value=0.9; out.connect(off.destination);
  ctx=off; master=out; OUT=null; ['click','notes','acc','bell'].forEach(k=>{ BUS[k]=off.createGain(); BUS[k].connect(out); });
  BUS.click.gain.value=1.15; BUS.notes.gain.value=1.1; BUS.acc.gain.value=1.1*accR;
  try{ log.forEach(e=>{ if(e.t<0||e.t>dur+0.5) return;
    if(which==='click'&&e.k==='c'){ S.clickSnd=e.snd; click(e.t,e.kind); }
    if(which==='notes'&&(e.k==='n'||e.k==='a')) snare(e.t,e.k==='a'); }); }
  finally{ ctx=sv.ctx; master=sv.master; OUT=sv.OUT; Object.assign(BUS,sv.bus); S.clickSnd=sv.snd; }
  return off.startRendering();
}
// where the camera file sits on the timeline: D = camera-file time minus timeline time (seconds)
function edPlace(meta,camAudio){
  const C=meta.cap, ev=(meta.log||[]).map(e=>e.t*1000);
  if(!C||C.t0==null||!C.L.length) return {D:0,how:'start'};
  const t0=Math.round(C.t0*1000), L=Float32Array.from(C.L,x=>x||0), R=Float32Array.from(C.R,x=>x||0);
  // 1. the capture runs a little behind the clock: find by how much using the app's own sounds (left)
  let lam=lagToTimes(onsetsOf(L),ev.map(t=>t-t0),-100,400); if(lam==null) lam=Math.round(2048/C.sr*1000);
  if(!camAudio||!C.mic) return {D:0,how:'start',lam};
  // 2. the mic heard on the clock (right) vs the mic inside the camera file
  const n=Math.max(0,R.length-lam+t0), mc=new Float32Array(n); for(let T=0;T<n;T++){ const k=T-t0+lam; mc[T]=k>=0&&k<R.length?R[k]:0; }
  const D=matchShift(onsetsOf(mc),onsetsOf(envOf(camAudio)),3000);
  return D==null?{D:0,how:'start',lam}:{D:D/1000,how:'mic',lam};
}

// start every track at timeline position t, at audio time `at`
function edSrc(c,buf,dest,pos,at){ if(!buf||pos>=buf.duration) return null; const s=c.createBufferSource(); s.buffer=buf; s.connect(dest); if(pos>=0) s.start(at,pos); else s.start(at-pos,0); return s; }

// keep a video on a target time without jumping: small drift is pulled in by speed, large drift by a seek

  const c=ED.cam, m=ED.meta, ar=m.camW&&m.camH?m.camH/m.camW:4/3;

}

/* preview loop */
function edLoop(){
  if(!ED||$('fEd').hidden) return;
  const vb=$('edBase'), vc=$('edCam'), cv=$('edCv'), t=edNow();
  if(ED.playing){ edSlave(vb,t); if(ED.hasCam) edSlave(vc,edCamT(t)); if(t>=ED.trim.b-0.03) edPause(); }
  else if(ED.hasCam&&vc.readyState>=2&&Math.abs((vc.currentTime||0)-edCamT(t))>0.04) vc.currentTime=edCamT(t);
  edDraw(cv.getContext('2d'),cv.width,cv.height,t,true);
  $('edCur').style.left=(100*clamp(t/ED.dur,0,1))+'%'; $('edPlay').hidden=ED.playing;
  if(ED.tool==='trim') edTrimInfo();
  ED.raf=requestAnimationFrame(edLoop);
}
