var ED=null, ED_SRC=null, ED_DEST=null, ED_PREV=null;
function edReady(v){ return new Promise(res=>{
  const done=()=>{ v.removeEventListener('timeupdate',fix); res(); };
  // recorded webm has no length; jumping far ahead makes the browser work it out
  const fix=()=>{ if(isFinite(v.duration)){ v.currentTime=0; v.removeEventListener('timeupdate',fix); setTimeout(res,60); } };
  if(v.readyState>=1){ if(isFinite(v.duration)) return res(); v.addEventListener('timeupdate',fix); v.currentTime=1e101; return; }
  v.addEventListener('loadedmetadata',()=>{ if(isFinite(v.duration)) done(); else { v.addEventListener('timeupdate',fix); v.currentTime=1e101; } },{once:true});
  setTimeout(res,2500); }); }
const relY=id=>{ const f=$('fFrame').getBoundingClientRect(), r=$(id).getBoundingClientRect(); return r.height?((r.top+r.height/2)-f.top)/f.height:null; };
async function edOpen(base,cam,meta){
  const hasCam=!!cam;
  if(ED){ URL.revokeObjectURL(ED.urlB); if(ED.urlC) URL.revokeObjectURL(ED.urlC); }
  ED={bBlob:base,cBlob:cam,meta,dur:meta.dur,urlB:URL.createObjectURL(base),urlC:cam?URL.createObjectURL(cam):null,
    cam:{x:meta.cam?meta.cam.x:0.04,y:meta.cam?meta.cam.y:0.6,w:meta.cam?meta.cam.w:0.33,shape:'rect',frame:true,mirror:true,pulse:false},
    lay:hasCam?'free':'none',hasCam,
    title:{on:!!S.fmTitle,text:S.fmTitle||'',y:relY('fChal')||0.2},handle:{on:!!S.fmHandle,text:S.fmHandle||'',y:relY('fHandle')||0.09},endT:{on:!!S.fmEnd,text:S.fmEnd||''},
    trim:{a:0,b:meta.dur},tool:hasCam?'cam':'txt',playing:false,drag:null,ptrs:new Map(),raf:0,guide:false};
  const vb=$('edBase'), vc=$('edCam'); vb.src=ED.urlB; vc.src=ED.urlC||'';
  if(!ED_SRC){ try{ ED_SRC=ctx.createMediaElementSource(vb); ED_DEST=ctx.createMediaStreamDestination(); ED_PREV=ctx.createGain(); ED_SRC.connect(ED_PREV).connect(ctx.destination); ED_SRC.connect(ED_DEST); }catch(e){} }
  $('fEd').hidden=false; $('edTools').querySelector('[data-tool="cam"]').disabled=!hasCam;
  await Promise.all([edReady(vb),hasCam?edReady(vc):null]);
  if(isFinite(vb.duration)&&vb.duration>0){ ED.dur=vb.duration; ED.trim.b=ED.dur; }
  // phones start the recorded sound a little after the picture: find the clicks in the file and measure by how much
  ED.abuf=null; ED.mic=null; ED.lag=0; ED.hp=0;
  try{ const all=await ctx.decodeAudioData(await base.arrayBuffer());
    // left = the app's sound, right = the microphone (recorded apart so headphone delay can be undone)
    if(all.numberOfChannels>=2){ const one=ch=>{ const b=ctx.createBuffer(1,all.length,all.sampleRate); b.copyToChannel(all.getChannelData(ch),0); return b; }; ED.abuf=one(0); ED.mic=one(1); ED.hp=S.hpLag||0; }
    else ED.abuf=all;
    ED.lag=edLag(ED.abuf,meta.beats,0); }catch(e){ ED.abuf=null; ED.mic=null; }
  vb.muted=!!ED.abuf;
  if(S.fmLay&&ED.hasCam){ if(S.fmLay==='none') ED.lay='none'; else edLayout(S.fmLay); }
  vb.currentTime=0; if(hasCam) vc.currentTime=0;
  edTool(ED.tool); edLoop();
}

/* sound sync */
// the clicks are at known times (meta.beats); slide them over the sound's loudness until they line up best
function edLag(buf,beats,shift){
  const bs=(beats||[]).map(b=>b-(shift||0)).filter(b=>b>=0).slice(0,16); if(bs.length<2) return 0;
  const d=buf.getChannelData(0), sr=buf.sampleRate, n=Math.floor(d.length/sr*1000), env=new Float32Array(n);
  for(let i=0;i<n;i++){ let m=0; const a=Math.floor(i*sr/1000), z=Math.min(d.length,a+Math.floor(sr/1000)); for(let k=a;k<z;k++){ const v=Math.abs(d[k]); if(v>m) m=v; } env[i]=m; }
  const on=new Float32Array(n); for(let i=4;i<n;i++) on[i]=Math.max(0,env[i]-env[i-4]); // rises in loudness = hits
  let best=0, bl=0;
  for(let L=-300;L<=1500;L+=2){ let s=0; for(const b of bs){ const c=Math.round(b*1000)+L; let m=0; for(let j=c-4;j<=c+4;j++) if(j>=0&&j<n&&on[j]>m) m=on[j]; s+=m; } if(s>best){ best=s; bl=L; } }
  return bl/1000;
}

/* geometry */
function edCamRect(W,H){
  const c=ED.cam, m=ED.meta, ar=m.camW&&m.camH?m.camH/m.camW:4/3;
  if(ED.lay==='half') return {x:0,y:0,w:W,h:H*0.47,shape:'fill'};
  const w=c.w*W, h=c.shape==='circle'?w:c.shape==='rect'?w*4/3:w*ar;
  return {x:c.x*W,y:c.y*H,w,h,shape:c.shape};
}

function edBpmAt(t){ let r=ED.meta.rounds[0]; ED.meta.rounds.forEach(x=>{ if(x.t<=t+0.05) r=x; }); return r?r.bpm:S.bpm; }
function wrapText(g,text,maxW){ const words=text.split(/\s+/), lines=[]; let l='';
  words.forEach(w=>{ const t=l?l+' '+w:w; if(g.measureText(t).width>maxW&&l){ lines.push(l); l=w; } else l=t; }); if(l) lines.push(l); return lines; }
function edText(g,W,H,text,cy,size,font,color,weight,dir){
  g.save(); g.font=`${weight||400} ${size}px ${font}`; g.fillStyle=color; g.textAlign='center'; g.textBaseline='middle'; g.direction=dir||'rtl';
  const lines=wrapText(g,text,W*0.86), lh=size*1.28, y0=cy-(lines.length-1)*lh/2; lines.forEach((l,i)=>g.fillText(l,W/2,y0+i*lh)); g.restore();
  return {top:y0-lh/2,bottom:y0+(lines.length-0.5)*lh}; }
function edDraw(g,W,H,t,preview){
  const vb=$('edBase'), vc=$('edCam'), m=ED.meta;

    if(ED.lay==='half'){ const VW=vb.videoWidth, VH=vb.videoHeight, sl=(m.scoreL!=null?m.scoreL:0.05), sr=(m.scoreR!=null?m.scoreR:0.95), st=m.scoreTop, sb=(m.scoreBot!=null?m.scoreBot:0.97);
      const sx=sl*VW, sy=st*VH, sw=(sr-sl)*VW, sh=(sb-st)*VH, top=H*0.555, ah=H*0.975-top, aw=W*0.96, sc=Math.min(aw/sw,ah/sh), dw=sw*sc, dd=sh*sc;
      g.drawImage(vb,sx,sy,sw,sh,(W-dw)/2,top+(ah-dd)/2,dw,dd); }
    else g.drawImage(vb,0,0,W,H);
  }
  const DF='"Secular One","Assistant",sans-serif';
  ED.hit={};
  if(ED.lay!=='half'){
    if(ED.title.on&&ED.title.text) ED.hit.title=edText(g,W,H,ED.title.text.replace(/\{BPM\}/gi,edBpmAt(t)),ED.title.y*H,W*0.066,DF,'#eef0f2');
    if(ED.handle.on&&ED.handle.text) ED.hit.handle=edText(g,W,H,ED.handle.text,ED.handle.y*H,W*0.036,'Assistant,sans-serif','#8e98a8',600,/^[\x00-\x7f]*$/.test(ED.handle.text)?'ltr':'rtl');
  } else {
    // half screen: the title and the level sit on a black band under the camera
    let r=m.rounds[0]; m.rounds.forEach((x,i)=>{ if(x.t<=t+0.05) r=x; }); const ri=Math.max(1,m.rounds.indexOf(r)+1), rt=Math.max(m.rounds_total||1,m.rounds.length);
    if(ED.title.on&&ED.title.text) ED.hit.title=edText(g,W,H,ED.title.text.replace(/\{BPM\}/gi,edBpmAt(t)),H*0.497,W*0.054,DF,'#eef0f2');

    g.save(); g.beginPath(); if(r.shape==='circle') g.arc(r.x+r.w/2,r.y+r.h/2,r.w/2,0,Math.PI*2); else rrP(g,r.x,r.y,r.w,r.h,rad); g.clip();

    g.translate(r.x+r.w/2,r.y+r.h/2); if(ED.cam.mirror) g.scale(-1,1); g.drawImage(vc,-dw/2,-dh/2,dw,dh); g.restore();
    if(ED.cam.frame&&r.shape!=='fill'){ let lw=W*0.006, glow=0;
      if(ED.cam.pulse){ const b=m.beats.find(x=>t>=x&&t-x<0.12); if(b!=null){ glow=1-(t-b)/0.12; lw*=1+glow*1.4; } }
      g.save(); g.strokeStyle='#ff9f1c'; g.lineWidth=lw; if(glow){ g.shadowColor='#ff9f1c'; g.shadowBlur=W*0.04*glow; }
      g.beginPath(); if(r.shape==='circle') g.arc(r.x+r.w/2,r.y+r.h/2,r.w/2-lw/2,0,Math.PI*2); else rrP(g,r.x+lw/2,r.y+lw/2,r.w-lw,r.h-lw,Math.max(0,rad-lw/2)); g.stroke(); g.restore(); }

    if(ED.handle.on&&ED.handle.text) pill(W*0.04,H*0.025,[[ED.handle.text,'#0b0c0e']],'left');
    pill(W*0.96,H*0.025,[['Chart','#0b0c0e'],['Ready','#e8870c']],'right');

/* preview loop */
function edLoop(){
  if(!ED||$('fEd').hidden) return;
  const vb=$('edBase'), vc=$('edCam'), cv=$('edCv'), t=vb.currentTime||0;
  if(ED.hasCam&&vc.readyState>=2&&Math.abs((vc.currentTime||0)-edCamT(t))>(ED.playing?0.15:0.04)) vc.currentTime=edCamT(t);
  if(ED.playing&&t>=ED.trim.b-0.03) edPause();
  edDraw(cv.getContext('2d'),cv.width,cv.height,t,true);
  $('edCur').style.left=(100*clamp(t/ED.dur,0,1))+'%'; $('edPlay').hidden=ED.playing;
  if(ED.tool==='trim') edTrimInfo();
  ED.raf=requestAnimationFrame(edLoop);
}

function edPause(){ if(!ED) return; ED.playing=false; $('edBase').pause(); $('edCam').pause(); edAudioStop(); }
function edSeek(t){ const vb=$('edBase'); vb.currentTime=clamp(t,0,ED.dur); if(ED.hasCam) $('edCam').currentTime=edCamT(vb.currentTime); if(ED.playing&&ED.abuf) edAudioStart(vb.currentTime+ED.lag,ctx.destination); }

async function edExport(){

  const vb=$('edBase'), vc=$('edCam'), cv=document.createElement('canvas'); cv.width=1080; cv.height=1920; const g=cv.getContext('2d');

  const tracks=[...cv.captureStream(30).getVideoTracks(),...(ED_DEST?ED_DEST.stream.getAudioTracks():[])];
  const mt=pickType(REC_TYPES); let rec; try{ rec=new MediaRecorder(new MediaStream(tracks),mt?{mimeType:mt,videoBitsPerSecond:8e6}:{videoBitsPerSecond:8e6}); }catch(e){ toast('לא הצלחתי לייצר את הסרטון'); return; }
  const chunks=[]; EDX={rec,cancel:false}; $('edExp').hidden=false; $('edPct').textContent='0%'; $('edRing').style.setProperty('--p',0);
  rec.ondataavailable=e=>{ if(e.data&&e.data.size) chunks.push(e.data); };
  rec.onstop=()=>{ const x=EDX; EDX=null; $('edExp').hidden=true; if(ED_PREV) ED_PREV.gain.value=1; vb.pause(); vc.pause(); if(!x||x.cancel) return;
    edAudioStop();
    const blob=new Blob(chunks,{type:(rec.mimeType||mt||'video/webm').split(';')[0]});
    // check the new file too: the recorder may delay the sound again; remember the correction for next time
    if(x.R!=null&&ED.abuf) blob.arrayBuffer().then(a=>ctx.decodeAudioData(a)).then(b=>{ const e=edLag(b,ED.meta.beats,ED.trim.a); if(Math.abs(e)<1.2){ S.recLag=+(x.R+e).toFixed(3); save(); } }).catch(()=>{}); if(FILM.url) URL.revokeObjectURL(FILM.url); FILM.blob=blob; FILM.url=URL.createObjectURL(blob);
    edClose(); $('fResVid').src=FILM.url; $('fResHint').textContent=''; $('fRes').hidden=false; };
  if(ED_PREV) ED_PREV.gain.value=0;
  vb.currentTime=ED.trim.a; if(ED.hasCam) vc.currentTime=edCamT(ED.trim.a);
  await new Promise(r=>{ vb.addEventListener('seeked',r,{once:true}); setTimeout(r,800); });
  const R=(S.recLag!=null?S.recLag:ED.lag); EDX.R=R;
  try{ await vb.play(); if(ED.hasCam) await vc.play(); }catch(e){}
  rec.start(500); if(ED.abuf&&ED_DEST) edAudioStart(vb.currentTime+ED.lag+R,ED_DEST); const len=ED.trim.b-ED.trim.a;
  const tick=()=>{ if(!EDX) return; const t=vb.currentTime, f=clamp((t-ED.trim.a)/len,0,1);
    if(ED.hasCam&&Math.abs(vc.currentTime-edCamT(t))>0.15) vc.currentTime=edCamT(t);

    if(t>=ED.trim.b-0.02||vb.ended){ try{ rec.stop(); }catch(e){} return; }
    requestAnimationFrame(tick); };

}
