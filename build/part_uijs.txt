/* ---------- wheels: a sideways scroller that always centres the chosen value ---------- */
function wheel(el,vals,onPick,fmtv){
  el.innerHTML=`<div class="wtrack">${vals.map(v=>`<button data-v="${v}">${fmtv?fmtv(v):v}</button>`).join('')}</div>`;
  const tr=el.firstChild; let cur=null, st=null, raf=0;
  const W={el,vals,
    set(v,anim){ cur=v; tr.querySelectorAll('button').forEach(b=>b.classList.toggle('on',+b.dataset.v===+v)); W.center(anim); },
    center(anim){ const b=tr.querySelector(`[data-v="${cur}"]`); if(!b||!el.clientWidth) return; const x=b.offsetLeft+b.offsetWidth/2-el.clientWidth/2; el.scrollTo({left:x,behavior:anim?'smooth':'auto'}); }};
  // after a fling, the value nearest the middle wins
  el.addEventListener('scroll',()=>{ clearTimeout(st); st=setTimeout(()=>{ const mid=el.scrollLeft+el.clientWidth/2; let best=null, bd=1e9;
    tr.querySelectorAll('button').forEach(b=>{ const d=Math.abs(b.offsetLeft+b.offsetWidth/2-mid); if(d<bd){ bd=d; best=b; } });
    if(best&&+best.dataset.v!==+cur){ cur=+best.dataset.v; W.set(cur,true); onPick(cur); uiTick(); } else if(best&&bd>2) W.center(true); },110); },{passive:true});
  tr.onclick=e=>{ const b=e.target.closest('button'); if(!b) return; cur=+b.dataset.v; W.set(cur,true); onPick(cur); uiTick(); };
  return W;
}

/* ---------- press and hold repeats, faster and faster ---------- */
function holdRepeat(root,sel,fn){
  let t=null, n=0; const stop=()=>{ clearTimeout(t); t=null; };
  root.addEventListener('pointerdown',e=>{ const b=e.target.closest(sel); if(!b) return; e.preventDefault(); n=0; fn(b,0); const go=()=>{ n++; fn(b,n); t=setTimeout(go,n<5?220:n<15?90:40); }; t=setTimeout(go,450); });
  ['pointerup','pointercancel','pointerleave'].forEach(ev=>root.addEventListener(ev,stop));
}
