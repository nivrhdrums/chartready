const PY_DEF={bpm:90,P:4,C:3,n:2,lay:{c:1,p:1,s:1},cnt:'c',pb:4,pm:'both',subs:[1,2,3,4],tStep:0,tMax:120};

const PY_SUBS=[[0.5,'חצאים'],[1,'רבעים'],[2,'שמיניות'],[3,'טריולות'],[4,'שש־עשריות'],[5,'קווינטולות'],[6,'סקסטולות']];
const PY_SHORT={0.5:'חצאים',1:'רבעים',2:'שמיניות',3:'טריולות',4:'שש־עשריות',5:'קווינטולות',6:'סקסטולות'};
const PY_RAT=[[4,3],[3,2],[2,3],[3,4],[5,4],[5,3],[4,5],[7,4]];
const PY_STG=[{L:{c:1,p:1,s:1},t:'הכל דולק',d:'קליק + פולס + חלוקה'},{L:{c:1,p:1,s:0},t:'בלי חלוקה',d:'מנגנים את החלוקה לבד'},{L:{c:1,p:.3,s:0},t:'פולס חלש',d:'הפולס בשקט, בקושי שומעים'},{L:{c:1,p:0,s:0},t:'רק הקליק',d:'המבחן האמיתי'}];

function pyS(){ if(!S.py||typeof S.py!=='object') S.py={}; const o=S.py; for(const k in PY_DEF) if(o[k]===undefined||o[k]===null) o[k]=JSON.parse(JSON.stringify(PY_DEF[k])); if(!pySubOk(o.n,o.P)) o.n=1; return o; }
const pySubOk=(n,P)=>n>=1||P%2===0;
const pyN=(n,P)=>Math.round(P*n);
const pyPulseBpm=o=>Math.round(o.bpm*o.P/o.C*10)/10;
const pyFmt=v=>String(Math.round(v*10)/10);
function pySubsList(){ const o=pyS(); const L=PY_SUBS.map(x=>x[0]).filter(n=>o.subs.includes(n)&&pySubOk(n,o.P)); return L.length?L:[o.n]; }

function pyOut(){ if(!PY.bus){ PY.bus=ctx.createGain(); PY.bus.gain.value=1; PY.bus.connect(master); } return PY.bus; }
function pyKeep(n){ PY.nodes.push(n); if(PY.nodes.length>400) PY.nodes.splice(0,200); }
function pyBell(t,v){ // קאובל לפולס
  const f=ctx.createBiquadFilter(), g=ctx.createGain(); f.type='bandpass'; f.frequency.value=800; f.Q.value=1.1;
  g.gain.setValueAtTime(0.0001,t); g.gain.exponentialRampToValueAtTime(0.55*v,t+0.002); g.gain.exponentialRampToValueAtTime(0.0001,t+0.17);
  f.connect(g).connect(pyOut());
  for(const fr of [562,836]){ const o=ctx.createOscillator(); o.type='square'; o.frequency.value=fr; o.connect(f); o.start(t); o.stop(t+0.19); pyKeep(o); }
}
function pyRim(t,v){ // רימשוט עדין לחלוקה
  const n=ctx.createBufferSource(); n.buffer=noiseBuf; const f=ctx.createBiquadFilter(); f.type='bandpass'; f.frequency.value=3200; f.Q.value=3;
  const g=ctx.createGain(); g.gain.setValueAtTime(0.42*v,t); g.gain.exponentialRampToValueAtTime(0.0001,t+0.03);
  n.connect(f).connect(g).connect(pyOut()); n.start(t); n.stop(t+0.05); pyKeep(n);
  const o=ctx.createOscillator(), og=ctx.createGain(); o.type='triangle'; o.frequency.value=1750;
  og.gain.setValueAtTime(0.0001,t); og.gain.exponentialRampToValueAtTime(0.14*v,t+0.001); og.gain.exponentialRampToValueAtTime(0.0001,t+0.025);
  o.connect(og).connect(pyOut()); o.start(t); o.stop(t+0.04); pyKeep(o);
}
function pyClick(t,acc){ const was=SCHED, n0=TRK.length; SCHED=true; try{ click(t,acc?'acc':'beat'); } finally{ SCHED=was; } const add=TRK.splice(n0); add.forEach(x=>pyKeep(x.n)); }

function pyNextInfo(){ // מה מתנגן בתיבה הבאה, ואז מתקדמים
  const o=pyS(), G=PY.g;
  if(!G.prog) return {prog:false,n:o.n,bpm:o.bpm};
  const subs=pySubsList(), pm=o.pm;
  const n=pm==='layers'?o.n:subs[G.si%subs.length];
  const stg=pm==='subs'?-1:G.stg;
  const L=pm==='subs'?{c:1,p:1,s:1}:PY_STG[G.stg].L;
  const segs=pm==='both'?4*subs.length:pm==='layers'?4:subs.length;
  const seg=pm==='both'?G.si*4+G.stg:pm==='layers'?G.stg:G.si;
  const left=o.pb-G.bar;
  let next;
  if(pm==='subs') next=PY_SUBS.find(x=>x[0]===subs[(G.si+1)%subs.length])[1];
  else if(G.stg<3) next=PY_STG[G.stg+1].t;
  else if(pm==='both') next=PY_SUBS.find(x=>x[0]===subs[(G.si+1)%subs.length])[1]+' · הכל דולק';
  else next='מההתחלה';
  const info={prog:true,n,bpm:G.bpm,L,stg,seg,segs,left,next,frac:G.bar/o.pb};
  // התקדמות
  const round=()=>{ if(o.tStep>0) G.bpm=Math.min(Math.max(o.tMax,o.bpm),G.bpm+o.tStep); };
  G.bar++;
  if(G.bar>=o.pb){ G.bar=0;
    if(pm==='subs'){ G.si++; if(G.si>=subs.length){ G.si=0; round(); } }
    else { G.stg++; if(G.stg>=4){ G.stg=0; if(pm==='both'){ G.si++; if(G.si>=subs.length){ G.si=0; round(); } } else round(); } }
  }
  return info;
}
function pyBar(t0){
  const o=pyS(), st=pyNextInfo(), dur=o.C*60/st.bpm, N=pyN(st.n,o.P), b=PY.barI++;
  const ev=[];
  for(let i=0;i<o.C;i++) ev.push({t:t0+dur*i/o.C,k:'c',i,st});
  for(let i=0;i<o.P;i++) ev.push({t:t0+dur*i/o.P,k:'p',i,st});
  for(let i=0;i<N;i++) ev.push({t:t0+dur*i/N,k:'s',i,st});
  ev.sort((a,b)=>a.t-b.t); PY.q.push(...ev);
  PY.bars.push({t0,dur,i:b,st,P:o.P,C:o.C,N}); if(PY.bars.length>6) PY.bars.shift();
  return dur;
}
function pyGain(e){ const o=pyS(); return e.st.prog?e.st.L[e.k]:(o.lay[e.k]?1:0); }
function pySound(e){ const g=pyGain(e); if(!g) return;
  if(e.k==='c') pyClick(e.t,e.i===0);
  else if(e.k==='p') pyBell(e.t,g*(e.i===0?1:0.8));
  else pyRim(e.t,g*(e.i===0?1:0.85));
}
function pyTick(){ if(!PY.play||!ctx) return; const now=ctx.currentTime;
  let guard=0; while(PY.nextT<now+0.35&&guard++<4) PY.nextT+=pyBar(PY.nextT);
  while(PY.q.length&&PY.q[0].t<now+0.12){ const e=PY.q.shift(); if(e.t<now-0.03) continue; pySound(e); }
}
async function pyStart(prog){
  if(typeof playing!=='undefined'&&playing) stop();
  await initAudio(); if(typeof applyVol==='function') applyVol();
  const o=pyS();
  PY.progOn=!!prog; PY.q=[]; PY.bars=[]; PY.barI=0; PY.shown=-1;
  PY.g={prog:PY.progOn,stg:0,si:0,bar:0,bpm:o.bpm};
  PY.nextT=ctx.currentTime+0.15; PY.play=true;
  clearInterval(PY.timer); PY.timer=setInterval(pyTick,25); pyTick();
  try{ if(navigator.wakeLock) PY.wake=await navigator.wakeLock.request('screen'); }catch(e){}
  pyRender();
}
function pyStop(){
  PY.play=false; clearInterval(PY.timer); PY.timer=0; PY.q=[];
  const t=ctx?ctx.currentTime:0; PY.nodes.forEach(n=>{ try{ n.stop(t); }catch(e){} }); PY.nodes=[];
  try{ PY.wake&&PY.wake.release(); }catch(e){} PY.wake=null;
  pyRender();
}
const pyToggle=()=>PY.play?pyStop():pyStart(PY.view==='prog');

function pyNoteHtml(n){ const o=pyS(), N=pyN(n,o.P), C=o.C, g=C*4, nm=PY_SUBS.find(x=>x[0]===n)[1];
  let on=0; for(let i=0;i<N;i++){ const x=i*g/N; if(Math.abs(x-Math.round(x))<1e-6) on++; }
  const head=`<b>${nm} על הפולס = ${N} על ${C}.</b> `;
  const PER={1:'רבעים',2:'שמיניות',3:'טריולות',4:'שש־עשריות',6:'סקסטולות',8:'32־יות'};
  if(N%C===0) return head+`<span class="ok">✓ חוזר לגריד:</span> בדיוק ${PER[N/C]||(N/C+' בכל קליק')} של הקליק.`;
  const sp=g/N, WR={3:'שמיניות מנוקדות',6:'רבעים מנוקדים',1.5:'שש־עשריות מנוקדות',12:'חצאים מנוקדים'};
  const wr=WR[Math.round(sp*100)/100]?` (בכתיבה: ${WR[Math.round(sp*100)/100]})`:'';
  if(on===N) return head+`כל הצלילים על השש־עשריות של הקליק${wr}.`;
  if(on===1) return head+`חוץ מה־1, כל הצלילים נופלים בין השש־עשריות של הקליק.`;
  return head+`רק ${on} מתוך ${N} נופלים על השש־עשריות של הקליק, השאר באמצע${wr}.`;
}

function pyFrame(){ if(!PY.on) return; PY.raf=requestAnimationFrame(pyFrame);
  if(!PY.play||!ctx||!PY.bars.length) return;
  const now=ctx.currentTime-(ctx.outputLatency||0)-(ctx.baseLatency||0);
  let b=null; for(const x of PY.bars) if(x.t0<=now) b=x; if(!b) return;
  if(b.i!==PY.shown) pyShowBar(b);
  const fr=Math.min(0.9999,(now-b.t0)/b.dur), el=$('pyLanes'), tr=el.querySelector('.pytr'), hd=$('pyHd');
  if(tr&&hd){ hd.style.opacity=1; hd.style.left=(tr.offsetLeft+tr.offsetWidth*(0.03+0.94*fr))+'px'; }
  const cnt={c:b.C,p:b.P,s:b.N};
  for(const k of ['c','p','s']){ const n=cnt[k], x=fr*n, i=Math.floor(x), since=(x-i)*b.dur/n, M=PY.marks&&PY.marks[k]; if(!M) continue;
    M.forEach((m,j)=>m.classList.toggle('hit',j===i&&since<0.11)); }
  const pi=Math.floor(fr*b.P), ci=Math.floor(fr*b.C);
  [...$('pyPulses').children].forEach((d,j)=>d.classList.toggle('on',j===pi));
  [...$('pyClicks').children].forEach((d,j)=>d.classList.toggle('on',j===ci));
}
function pyKey(e){ if(/INPUT|SELECT|TEXTAREA/.test(e.target.tagName)) return; if($('kpWrap').classList.contains('open')) return;
  if(e.code==='Space'){ e.preventDefault(); pyToggle(); }
  else if(e.key==='Escape'){ e.preventDefault(); if(!$('pySheet').hidden) $('pySheet').hidden=true; else $('pyX').click(); }
  else if(e.key==='ArrowUp'||e.key==='ArrowRight'){ e.preventDefault(); pySetBpm(pyS().bpm+(e.shiftKey?5:1)); }
  else if(e.key==='ArrowDown'||e.key==='ArrowLeft'){ e.preventDefault(); pySetBpm(pyS().bpm-(e.shiftKey?5:1)); }
}
